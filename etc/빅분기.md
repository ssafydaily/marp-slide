## 이진 분류
```python
import pandas as pd

# 1) 데이터 불러오기
#    - train: 정답(y)이 들어있는 학습용 데이터 (y=1 가입 / y=0 미가입)
#    - test : 정답이 없는 데이터 → 우리가 "가입 확률"을 예측해서 제출
train = pd.read_csv('data/bank_train.csv')
test = pd.read_csv('data/bank_test.csv')

# 2) 정답(target) 분리
#    pop(): train에서 정답 컬럼 'y'를 꺼내 따로 보관
target = train.pop('y')

# 3) 범주형 변수 원-핫 인코딩
#    job, marital, education, contact, month, poutcome 등 글자 컬럼이 많음 → 0/1 숫자로 변환
#    train/test를 합쳐서 인코딩하면 컬럼이 항상 일치 (한쪽에만 있는 범주 문제 방지)
all_df = pd.concat([train, test])
all_df = pd.get_dummies(all_df)
train = all_df[:len(target)]
test = all_df[len(target):]
print(train.shape, test.shape)   # train/test 컬럼 수가 같은지 확인

# 4) 검증용 분리 (제출 전 성능 비교)
#    stratify=target: 불균형 데이터이므로 분할 시 가입 비율을 train/val에 동일하게 유지
from sklearn.model_selection import train_test_split
X_tr, X_val, y_tr, y_val = train_test_split(train, target, test_size=0.2, random_state=0, stratify=target)

# 5) 세 가지 모델 학습 (이진 분류 → Classifier)
#    ROC-AUC는 "확률"로 평가 → predict()가 아니라 predict_proba(...)[:, 1] 사용!
#    [:, 1] = 양성(가입=1)일 확률 컬럼
from sklearn.metrics import roc_auc_score

from sklearn.ensemble import RandomForestClassifier
rf = RandomForestClassifier(random_state=0)
rf.fit(X_tr, y_tr)
print('RF  ', roc_auc_score(y_val, rf.predict_proba(X_val)[:, 1]))

from lightgbm import LGBMClassifier
lgb = LGBMClassifier(random_state=0, verbose=-1)
lgb.fit(X_tr, y_tr)
print('LGBM', roc_auc_score(y_val, lgb.predict_proba(X_val)[:, 1]))

from xgboost import XGBClassifier
xgb = XGBClassifier(random_state=0, verbosity=0)
xgb.fit(X_tr, y_tr)
print('XGB ', roc_auc_score(y_val, xgb.predict_proba(X_val)[:, 1]))

# ──────────────────────────────────────────────────────────────
# [흔한 실수] 확률이 아니라 0/1 라벨을 제출하면?
#   rf.predict(X_val)는 0 또는 1만 내놓음 → ROC-AUC가 크게 떨어진다.
#   ROC-AUC는 "확률 순위"를 보는 지표라서 반드시 predict_proba 를 써야 함!
# ──────────────────────────────────────────────────────────────

# 6) 검증 AUC가 가장 높았던 모델을 전체 train으로 다시 학습 후 test 예측
#    중요: 제출은 정답 없는 test의 "가입 확률"을 예측해야 함
lgb.fit(train, target)
pred = lgb.predict_proba(test)[:, 1]   # 양성(가입) 확률

# 7) 제출 파일 저장 (pred 컬럼 1개만, index 제거)
result = pd.DataFrame({'pred': pred})
result.to_csv('result.csv', index=False)

# 제출파일 확인
print("\n ===== 제출파일 (샘플) =====")
print(pd.read_csv("result.csv").head())
print("\n ===== 제출파일 (크기 확인) =====")
print(pd.read_csv("result.csv").shape)

```
